\paragraph{Problem 1.} Let $X$ and $Y$ be independent with
$X, Y \sim \mathrm{Exp}(\lambda)$. Find the distribution of $\min(X, Y)$.

\paragraph{Problem 2.} Show that for any random variable $X$ with finite
variance and any $t > 0$,
\[
  \Pr\bigl(|X - \mathbb{E}X| \ge t\bigr) \le \frac{\operatorname{Var} X}{t^2}.
\]

\paragraph{Problem 3.} A fair die is rolled until a six appears. Compute the
expected number of rolls and its variance.

\paragraph{Problem 4.} Let $(X_n)$ be i.i.d.\ with mean $\mu$. Prove that
$\frac{1}{n}\sum_{k=1}^{n} X_k \to \mu$ in probability.
